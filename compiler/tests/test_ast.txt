int soma(int a, int b) {
    return a + b;
}

int multiplicar(int x, int y) {
    return x * y;
}

void mensagem() {
    return;
}

int main() {

    // =========================
    // DECLARAÇÕES VÁLIDAS
    // =========================

    int x = 10;
    int y = 5;
    int z;

    char letra = 'A';
    char quebra = '\n';

    // =========================
    // EXPRESSÕES VÁLIDAS
    // =========================

    x = x + y;
    x = x - y;
    x = x * y;
    x = x / y;
    x = z;

    x = (x + y) * 2;

    x = -x;
    x = !y;

    // =========================
    // CHAMADAS DE FUNÇÃO
    // =========================

    z = soma(x, y);
    z = multiplicar(z, 2);

    mensagem();

    // =========================
    // INCREMENTOS VÁLIDOS
    // =========================

    x++;
    x--;
    ++x;
    --x;

    // =========================
    // IF / ELSE VÁLIDOS
    // =========================

    if(x > y) {
        x = x + 1;
    }

    if(x > y && y > 0) {
        x = x + 1;
    } else {
        x = x - 1;
    }

    if(!x) {
        x = 0;
    }

    // =========================
    // WHILE VÁLIDO
    // =========================

    while(x > 0) {
        x--;
    }

    // =========================
    // FOR VÁLIDOS
    // =========================

    for(int i = 0; i < 10; i++) {
        z = z + i;
    }

    for(y = 0; y < 5; y = y + 1) {
        z = z + y;
    }

    // =========================
    // BREAK / CONTINUE VÁLIDOS
    // =========================

    while(z > 0) {
        z--;
        if(z == 8) {
            continue;
        }
        if(z == 3) {
            break;
        }
    }

    // STRINGS VÁLIDAS
